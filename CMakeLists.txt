cmake_minimum_required(VERSION 3.20.0)
set(SATELLITE_STAGE "status" CACHE STRING "display, ble, or status")
if(SATELLITE_STAGE STREQUAL "ble")
  list(APPEND EXTRA_CONF_FILE "${CMAKE_CURRENT_SOURCE_DIR}/config/ble.conf")
  list(APPEND EXTRA_CONF_FILE "${CMAKE_CURRENT_SOURCE_DIR}/config/bluetooth.conf")
elseif(SATELLITE_STAGE STREQUAL "display" OR SATELLITE_STAGE STREQUAL "status")
  list(APPEND EXTRA_CONF_FILE "${CMAKE_CURRENT_SOURCE_DIR}/config/lvgl.conf")
  if(SATELLITE_STAGE STREQUAL "display")
    list(APPEND EXTRA_CONF_FILE "${CMAKE_CURRENT_SOURCE_DIR}/config/display.conf")
  else()
    list(APPEND EXTRA_CONF_FILE "${CMAKE_CURRENT_SOURCE_DIR}/config/bluetooth.conf")
  endif()
else()
  message(FATAL_ERROR "SATELLITE_STAGE must be display, ble or status")
endif()
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(zmk_satellite_view VERSION 0.1.0 LANGUAGES C)
target_include_directories(app PRIVATE src)
target_sources(app PRIVATE src/main.c src/protocol.c src/state.c)
target_sources_ifdef(CONFIG_SATELLITE_BLE app PRIVATE src/scanner.c)
target_sources_ifdef(CONFIG_SATELLITE_DISPLAY app PRIVATE src/ui.c)
